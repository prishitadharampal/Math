\begin{problem}{1.3}
    Let $Y$ be the algebraic set in $\mathbb A^3$ defined by the two polynomials
    \[
    x^2-yz
    \qquad\text{and}\qquad
    xz-x.
    \]
    Show that $Y$ is a union of three irreducible components. Describe them and find their prime ideals.
\end{problem}

\begin{solution}
    \bbni
    Note that $xz -x = x(z -1)$ and $x^2 -yz$ is irreducible. So 
    $Y = \V(x^2 - yz, x, z-1)  = \V(x^2- yz) \cup \V(x) \cup \V(z-1)$. Their prime ideals are the ones they generate because they are irreducible. 
\end{solution}

\begin{problem}{1.7}
\begin{enumerate}
    \item[(a)] Show that the following conditions are equivalent for a topological space $X$:
    \begin{enumerate}
        \item[(i)] $X$ is noetherian;
        \item[(ii)] every nonempty family of closed subsets has a minimal element;
        \item[(iii)] $X$ satisfies the ascending chain condition for open subsets;
        \item[(iv)] every nonempty family of open subsets has a maximal element.
    \end{enumerate}

    \item[(b)] A noetherian topological space is \emph{quasi-compact}, i.e., every open cover has a finite subcover.

    \item[(c)] Any subset of a noetherian topological space is noetherian in its induced topology.

    \item[(d)] A noetherian space which is also Hausdorff must be a finite set with the discrete topology.
\end{enumerate}
\end{problem}

\begin{problem}{1.8}

\end{problem}

\begin{problem}{1.9}

\end{problem}

\begin{problem}{1.10}
\begin{enumerate}
    \item[(a)] If $Y$ is any subset of a topological space $X$, then $\dim Y\leq \dim X$.

    \item[(b)] If $X$ is a topological space which is covered by a family of open subsets $\{U_i\}$, then
    \[
    \dim X=\sup \dim U_i.
    \]

    \item[(c)] Give an example of a topological space $X$ and a dense open subset $U$ with
    \[
    \dim U<\dim X.
    \]

    \item[(d)] If $Y$ is a closed subset of an irreducible finite-dimensional topological space $X$, and if
    \[
    \dim Y=\dim X,
    \]
    then $Y=X$.

    \item[(e)] Give an example of a noetherian topological space of infinite dimension.
\end{enumerate}
\end{problem}
